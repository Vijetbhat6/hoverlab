    84→const slug = (s) => s.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '')

    86→const seq = () => String(++_seq).padStart(4, '0')

    89→const effects = []

    92→/**
    93→ * Build an effect object with normalized fields.
    94→ * `klass` is the unique CSS class to use (already generated by caller).
    95→ */

    97→  return {
    98→    id: slug(name) + '-' + seq(),
    99→    name,

   102→    html,
   103→    css,
   104→    tags,
   105→    darkSurface,
   106→    previewClass,
   107→    generated: true,
   108→  }
