    62→export function EffectDetail({ effect, similar, prev, next }: EffectDetailProps) {
    63→  const surfaceDark = effect.darkSurface
    64→  const previewRef = React.useRef<HTMLDivElement>(null)
    65→  const { has, toggle } = useFavorites()

    67→  const { has: hasBundle, toggle: toggleBundle } = useBundle()

    69→  const [bundleOpen, setBundleOpen] = React.useState(false)
    70→
    71→  /* Customization state -----------------------------------------------
    72→   * Initialize from the URL hash on first mount, so a shared link
    73→   * (e.g. /effect/btn-gradient#hue=15&sat=25) restores the exact
    74→   * customized view. When the user navigates to a different effect
    75→   * via prev/next, we re-read the hash for that URL too.
    76→   */
    77→  const [opts, setOpts] = React.useState<CustomizationOptions>(() =>
    78→    typeof window === 'undefined'
    79→      ? DEFAULT_CUSTOMIZATION
    80→      : hashToOpts(parseHash(window.location.hash)),
    81→  )
    82→  // Flag used to suppress the very first hash-write effect (so we don't
    83→  // re-write the hash that the user just landed on, which would trigger
    84→  // an unnecessary history entry).
    85→  const skipHashWriteRef = React.useRef(true)
    86→
    87→  React.useEffect(() => {
    88→    // When effect.id changes (prev/next navigation via internal Link),
    89→    // re-read the hash because Next.js client-side navigation does NOT
    90→    // re-mount this component — it just updates props.
    91→    const next = hashToOpts(parseHash(window.location.hash))
    92→    setOpts(next)
    93→    skipHashWriteRef.current = true
    94→  }, [effect.id])
    95→
    96→  // Write opts to the URL hash whenever they change (except on the very
    97→  // first run after a navigation, to avoid a redundant history entry).
    98→  React.useEffect(() => {
    99→    if (skipHashWriteRef.current) {
   100→      skipHashWriteRef.current = false
   101→      return

   103→    const hash = optsToHash(opts)
   104→    const target = hash ? `#${hash}` : ''
   105→    if (window.location.hash !== target) {
   106→      window.history.replaceState(null, '', window.location.pathname + window.location.search + target)
   107→    }

   109→
   110→  // Listen for hashchange events (back/forward, manual URL edits) and
   111→  // sync opts from the URL. This makes the share-link round-trip work.

   113→    const onHashChange = () => {
   114→      const next = hashToOpts(parseHash(window.location.hash))
   115→      skipHashWriteRef.current = true
   116→      setOpts(next)
   117→    }

   119→    return () => window.removeEventListener('hashchange', onHashChange)
   120→  }, [])
   121→
   122→  const customizedCss = React.useMemo(

   127→  const activePreset = matchingPreset(opts)
   128→
   129→  // Detail-page keyboard shortcuts:
   130→  //   j / →  → next effect
   131→  //   k / ←  → previous effect
   132→  //   f      → toggle favorite
   133→  //   s      → toggle bundle (saves current customization)

   135→  //   b      → open the bundle drawer
   136→  //   Esc    → close any open dialog (handled by Radix)
   137→  // We rely on Next.js <Link> for prev/next so client-side navigation
   138→  // preserves state. Pressing j/k triggers a click on the corresponding
   139→  // link element via ref.
   140→  const prevLinkRef = React.useRef<HTMLAnchorElement>(null)
   141→  const nextLinkRef = React.useRef<HTMLAnchorElement>(null)
   142→
   143→  React.useEffect(() => {
   144→    function isTypingTarget(t: EventTarget | null): boolean {
   145→      if (!(t instanceof HTMLElement)) return false
   146→      const tag = t.tagName
   147→      return tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || t.isContentEditable
   148→    }
   149→    function onKey(e: KeyboardEvent) {
   150→      if (e.metaKey || e.ctrlKey || e.altKey) return
   151→      if (isTypingTarget(e.target)) return
   152→      const key = e.key.toLowerCase()
   153→      if (key === 'j' || e.key === 'ArrowRight') {
   154→        e.preventDefault()
   155→        nextLinkRef.current?.click()
   156→      } else if (key === 'k' || e.key === 'ArrowLeft') {
   157→        e.preventDefault()
   158→        prevLinkRef.current?.click()
   159→      } else if (key === 'f') {
   160→        e.preventDefault()
   161→        toggle(effect.id)
   162→      } else if (key === 's') {
   163→        e.preventDefault()
   164→        toggleBundle(effect.id, opts)
   165→      } else if (key === 'c') {
   166→        e.preventDefault()
   167→        navigator.clipboard
   168→          .writeText(customizedCss)
   169→          .then(() => toast.success('Copied CSS to clipboard'))
   170→          .catch(() => toast.error('Copy failed — please copy manually'))

   172→        e.preventDefault()
   173→        // Toggle the drawer so `b` works as both "open" and "close".
   174→        setBundleOpen((v) => !v)
   175→      }
   176→    }
   177→    window.addEventListener('keydown', onKey)
   178→    return () => window.removeEventListener('keydown', onKey)

   236→          <span className="font-mono">{effect.id}</span>

   239→              <Sparkles className="h-2.5 w-2.5" /> Edited

   242→        </div>
   243→      </div>
   244→
   245→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-[1fr_320px]">

   479→/* ============================================================
   480→ *  Mini preview for the similar-effects rail.
   481→ *  Injects the effect's CSS scoped under a unique wrapper class
   482→ *  so multiple similar effects can render side by side.
   483→ * ========================================================== */
   484→
   485→let previewSeq = 0
   486→
   487→function SimilarPreview({ effect }: { effect: Effect }) {
   488→  const [wrapId] = React.useState(() => `fx-preview-${++previewSeq}`)
   489→  // Scope the CSS by prefixing every selector with our wrapper class.
   490→  // Cheap & cheerful — works for the simple class-based selectors we generate.
   491→  const scopedCss = React.useMemo(() => {
   492→    // Replace each selector group `selector1, selector2 {` with
   493→    // `.wrapId selector1, .wrapId selector2 {`
   494→    return effect.css.replace(/(^|\})\s*([^{}]+)\{/g, (_m, brace, selectors) => {
   495→      const scoped = selectors
   496→        .split(',')
   497→        .map((s: string) => `.${wrapId} ${s.trim()}`)
   498→        .join(', ')
   499→      return `${brace} ${scoped} {`
   500→    })
   501→  }, [effect.css, wrapId])
   502→
   503→  return (
   504→    <div
   505→      className={cn(
   506→        'flex h-10 w-10 shrink-0 items-center justify-center overflow-hidden rounded-md border border-border/40',
   507→        effect.darkSurface ? 'bg-slate-950' : 'bg-muted/40',
   508→      )}
   509→    >
   510→      <style dangerouslySetInnerHTML={{ __html: scopedCss }} />
   511→      <div
   512→        className={wrapId}
   513→        style={{ transform: 'scale(0.45)', transformOrigin: 'center' }}
   514→        dangerouslySetInnerHTML={{ __html: effect.html }}
   515→      />
   516→    </div>
   517→  )
   518→}

   529→  activePreset: Preset | null
   530→  isCustomized: boolean
   531→  customizedCss: string
   532→  combinedSnippet: string
   533→  onCopyShareLink: () => void

   536→function CustomizePanel({
   537→  opts,
   538→  onChange,
   539→  onReset,
   540→  onPreset,
   541→  activePreset,
   542→  isCustomized,
   543→  customizedCss,
   544→  combinedSnippet,
   545→  onCopyShareLink,
   546→}: CustomizePanelProps) {
   547→  const [copiedLink, setCopiedLink] = React.useState(false)
   548→
   549→  // Reset the "copied" indicator on the share button after a short delay.
   550→  // We don't trigger this here — the parent's onCopyShareLink fires the
   551→  // toast. We just flip the icon back after 1.6s when the button is hit.
   552→  React.useEffect(() => {
   553→    if (!copiedLink) return
   554→    const t = window.setTimeout(() => setCopiedLink(false), 1600)
   555→    return () => window.clearTimeout(t)
   556→  }, [copiedLink])
   557→
   558→  function handleShareClick() {
   559→    onCopyShareLink()
   560→    setCopiedLink(true)
   561→  }
   562→
   563→  return (
   564→    <div className="space-y-4 rounded-lg border border-border/60 bg-muted/20 p-3">
   565→      <p className="text-xs text-muted-foreground">
   566→        Slide to transform the live preview. The customized CSS is reflected
   567→        in real time — copy it, or share the link to lock in your tweak.
   568→      </p>
   569→
   570→      {/* Presets */}
   571→      <div className="space-y-2">
   572→        <div className="flex items-center justify-between">
   573→          <label className="text-xs font-semibold text-foreground">Presets</label>
   574→          {activePreset ? (
   575→            <span className="text-[11px] font-medium text-primary">{activePreset.name}</span>
   576→          ) : null}
   577→        </div>
   578→        <div className="flex flex-wrap gap-1.5">
   579→          {PRESETS.map((p) => {
   580→            const isActive = activePreset?.id === p.id
   581→            return (
   582→              <button
   583→                key={p.id}
   584→                type="button"
   585→                onClick={() => onPreset(p)}
   586→                title={p.description}
   587→                className={cn(
   588→                  'inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-[11px] font-medium transition-all',
   589→                  isActive
   590→                    ? 'border-primary bg-primary/10 text-primary'
   591→                    : 'border-border/60 bg-background/60 text-muted-foreground hover:border-primary/40 hover:text-foreground',
   592→                )}
   593→              >
   594→                <span
   595→                  className="h-2.5 w-2.5 rounded-full border border-black/10"
   596→                  style={{ backgroundColor: p.swatch }}
   597→                  aria-hidden="true"
   598→                />
   599→                {p.name}
   600→              </button>
   601→            )
   602→          })}
   603→        </div>
   604→      </div>
   605→
   606→      <div className="h-px bg-border/40" />
   607→
   608→      {/* Sliders */}
   609→      <SliderRow
   610→        label="Hue"
   611→        value={opts.hue}
   612→        min={-180}
   613→        max={180}
   614→        step={5}
   615→        unit="°"
   616→        onChange={(v) => onChange({ ...opts, hue: v })}
   617→        description="Rotate every color around the wheel."
   618→      />
   619→
   620→      <SliderRow
   621→        label="Saturation"
   622→        value={opts.saturation}
   623→        min={-100}
   624→        max={100}
   625→        step={5}
   626→        unit="%"
   627→        format={(v) => `${v > 0 ? '+' : ''}${v}%`}
   628→        onChange={(v) => onChange({ ...opts, saturation: v })}
   629→        description="Boost or mute color intensity. -100 = grayscale."
   630→      />
   631→
   632→      <SliderRow
   633→        label="Size"
   634→        value={opts.scale}
   635→        min={0.5}
   636→        max={1.5}
   637→        step={0.05}
   638→        unit="×"
   639→        format={(v) => `${v.toFixed(2)}×`}
   640→        onChange={(v) => onChange({ ...opts, scale: v })}
   641→        description="Scale every px/rem dimension."
   642→      />
   643→
   644→      <SliderRow
   645→        label="Speed"
   646→        value={opts.speed}
   647→        min={0.25}
   648→        max={3}
   649→        step={0.25}
   650→        unit="×"
   651→        format={(v) => `${v.toFixed(2)}×`}
   652→        onChange={(v) => onChange({ ...opts, speed: v })}
   653→        description="Multiply every animation duration."
   654→      />
   655→
   656→      {/* Action buttons */}
   657→      <div className="flex flex-wrap items-center gap-2 pt-1">
   658→        <Button
   659→          size="sm"
   660→          variant="outline"
   661→          onClick={onReset}
   662→          disabled={!isCustomized}
   663→          className="h-8 gap-1.5"

   665→          <RotateCcw className="h-3.5 w-3.5" /> Reset
   666→        </Button>
   667→        <Button
   668→          size="sm"
   669→          variant={isCustomized ? 'default' : 'outline'}

   679→              <Link2 className="h-3.5 w-3.5" /> Share link
   680→            </>
   681→          )}
   682→        </Button>
   683→        <CodeBlock
   684→          code={customizedCss}
   685→          filename="customized.css"
   686→          language="css"
   687→          extraCopy={{ label: 'Copy both', text: combinedSnippet }}
   688→        />
   689→      </div>
   690→    </div>
   691→  )
   692→}

   694→/* ============================================================
   695→ *  SliderRow — label + value badge + slider
   696→ * ========================================================== */
   697→
   698→interface SliderRowProps {
   699→  label: string
   700→  value: number
   701→  min: number
   702→  max: number
   703→  step: number
   704→  unit: string
   705→  format?: (v: number) => string
   706→  description?: string
   707→  onChange: (v: number) => void
   708→}
   709→
   710→function SliderRow({
   711→  label,
   712→  value,
   713→  min,
   714→  max,
   715→  step,
   716→  unit,
   717→  format,
   718→  description,
   719→  onChange,
   720→}: SliderRowProps) {
   721→  const display = format ? format(value) : `${value}${unit}`
   722→  return (
   723→    <div className="space-y-1.5">
   724→      <div className="flex items-center justify-between">
   725→        <label className="text-xs font-semibold text-foreground">{label}</label>
   726→        <span className="rounded-md bg-background px-1.5 py-0.5 font-mono text-[11px] text-muted-foreground tabular-nums">
   727→          {display}
   728→        </span>
   729→      </div>
   730→      <Slider
   731→        value={[value]}
   732→        min={min}
   733→        max={max}
   734→        step={step}
   735→        onValueChange={(arr) => arr[0] !== undefined && onChange(arr[0])}
   736→        className="w-full"
   737→      />
   738→      {description ? (
   739→        <p className="text-[11px] text-muted-foreground/80">{description}</p>
   740→      ) : null}
   741→    </div>
   742→  )
   743→}
