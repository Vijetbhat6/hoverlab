    44→    44→interface EffectDetailProps {
    45→    45→  effect: Effect
    46→    46→  /** Up to 6 similar effects (same category, excluding `effect`). */
    47→    47→  similar: Effect[]
    48→    48→  /** Previous effect in the catalog (for prev/next nav). Null if at start. */

    54→    54→/**
    55→    55→ * Full-page detail view for a single effect.
    56→    56→ *
    57→    57→ * - Large live preview (always visible, customizable in real time)
    58→    58→ * - Customize tab with hue / size / speed sliders
    59→    59→ * - Code tab with HTML + CSS (and "Copy both")
    60→    60→ * - Similar-effects rail (same category)
    61→    61→ * - Prev / next navigation across the catalog
    62→    62→ * - Favorite heart toggle (synced with the grid via localStorage events)
    63→    63→ */
    64→    64→export function EffectDetail({ effect, similar, prev, next }: EffectDetailProps) {
    65→    65→  const surfaceDark = effect.darkSurface
    66→    66→  const previewRef = React.useRef<HTMLDivElement>(null)
    67→    67→  const { has, toggle } = useFavorites()
    68→    68→  const isFav = has(effect.id)
    69→    69→  const { has: hasBundle, toggle: toggleBundle } = useBundle()
    70→    70→  const inBundle = hasBundle(effect.id)
    71→    71→  const [bundleOpen, setBundleOpen] = React.useState(false)
    72→    72→
    73→    73→  /* Active tab state ---------------------------------------------------
    74→    74→   * We control the Tabs (value + onValueChange) instead of using
    75→    75→   * defaultValue so we can auto-open the Customize tab when the user
    76→    76→   * arrives via a share link that contains customization in the URL
    77→    77→   * hash (e.g. /effect/btn-gradient#hue=15&sat=25). On subsequent
    78→    78→   * prev/next navigations the user's current tab is preserved — only
    79→    79→   * the initial mount checks the hash.
    80→    80→   *
    81→    81→   * We intentionally start with 'code' on both server and client
    82→    82→   * (avoids a hydration mismatch — the server can't see the URL hash)
    83→    83→   * and flip to 'customize' in a useEffect if the hash has any
    84→    84→   * customization params.
    85→    85→   */
    86→    86→  const [activeTab, setActiveTab] = React.useState<'code' | 'customize'>('code')
    87→    87→  React.useEffect(() => {
    88→    88→    const parts = parseHash(window.location.hash)
    89→    89→    const hasCustomization =
    90→    90→      parts.hue !== undefined ||
    91→    91→      parts.saturation !== undefined ||
    92→    92→      parts.scale !== undefined ||
    93→    93→      parts.speed !== undefined
    94→    94→    if (hasCustomization) setActiveTab('customize')
    95→    95→    // Only run on mount — prev/next nav preserves the user's tab.
    96→    96→  }, [])
    97→    97→
    98→    98→  /* Customization state -----------------------------------------------
    99→    99→   * Initialize from the URL hash on first mount, so a shared link
   100→   100→   * (e.g. /effect/btn-gradient#hue=15&sat=25) restores the exact
   101→   101→   * customized view. When the user navigates to a different effect
   102→   102→   * via prev/next, we re-read the hash for that URL too.
   103→   103→   */
   104→   104→  const [opts, setOpts] = React.useState<CustomizationOptions>(() =>
   105→   105→    typeof window === 'undefined'
   106→   106→      ? DEFAULT_CUSTOMIZATION
   107→   107→      : hashToOpts(parseHash(window.location.hash)),
   108→   108→  )
   109→   109→  // Flag used to suppress the very first hash-write effect (so we don't
   110→   110→  // re-write the hash that the user just landed on, which would trigger
   111→   111→  // an unnecessary history entry).
   112→   112→  const skipHashWriteRef = React.useRef(true)
   113→   113→
   114→   114→  React.useEffect(() => {
   115→   115→    // When effect.id changes (prev/next navigation via internal Link),
   116→   116→    // re-read the hash because Next.js client-side navigation does NOT
   117→   117→    // re-mount this component — it just updates props.
   118→   118→    const next = hashToOpts(parseHash(window.location.hash))
   119→   119→    setOpts(next)
   120→   120→    skipHashWriteRef.current = true
   121→   121→  }, [effect.id])
   122→   122→
   123→   123→  // Write opts to the URL hash whenever they change (except on the very
   124→   124→  // first run after a navigation, to avoid a redundant history entry).
   125→   125→  React.useEffect(() => {
   126→   126→    if (skipHashWriteRef.current) {
   127→   127→      skipHashWriteRef.current = false
   128→   128→      return
   129→   129→    }
   130→   130→    const hash = optsToHash(opts)
   131→   131→    const target = hash ? `#${hash}` : ''
   132→   132→    if (window.location.hash !== target) {
   133→   133→      window.history.replaceState(null, '', window.location.pathname + window.location.search + target)
   134→   134→    }
   135→   135→  }, [opts])
   136→   136→
   137→   137→  // Listen for hashchange events (back/forward, manual URL edits) and
   138→   138→  // sync opts from the URL. This makes the share-link round-trip work.
   139→   139→  React.useEffect(() => {
   140→   140→    const onHashChange = () => {
   141→   141→      const next = hashToOpts(parseHash(window.location.hash))
   142→   142→      skipHashWriteRef.current = true
   143→   143→      setOpts(next)
   144→   144→    }
   145→   145→    window.addEventListener('hashchange', onHashChange)
   146→   146→    return () => window.removeEventListener('hashchange', onHashChange)
   147→   147→  }, [])
   148→   148→
   149→   149→  const customizedCss = React.useMemo(
   150→   150→    () => customizeCss(effect.css, opts),
   151→   151→    [effect.css, opts],
   152→   152→  )
   153→   153→  const isCustomized = customizedCss !== effect.css
   154→   154→  const activePreset = matchingPreset(opts)
   155→   155→
   156→   156→  // Detail-page keyboard shortcuts:
   157→   157→  //   j / →  → next effect
   158→   158→  //   k / ←  → previous effect
   159→   159→  //   f      → toggle favorite
   160→   160→  //   s      → toggle bundle (saves current customization)
   161→   161→  //   c      → copy current (customized) CSS
   162→   162→  //   b      → open the bundle drawer
   163→   163→  //   Esc    → close any open dialog (handled by Radix)
   164→   164→  // We rely on Next.js <Link> for prev/next so client-side navigation
   165→   165→  // preserves state. Pressing j/k triggers a click on the corresponding
   166→   166→  // link element via ref.
   167→   167→  const prevLinkRef = React.useRef<HTMLAnchorElement>(null)
   168→   168→  const nextLinkRef = React.useRef<HTMLAnchorElement>(null)
   169→   169→
   170→   170→  React.useEffect(() => {
   171→   171→    function isTypingTarget(t: EventTarget | null): boolean {
   172→   172→      if (!(t instanceof HTMLElement)) return false
   173→   173→      const tag = t.tagName
   174→   174→      return tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || t.isContentEditable
   175→   175→    }
   176→   176→    function onKey(e: KeyboardEvent) {
   177→   177→      if (e.metaKey || e.ctrlKey || e.altKey) return
   178→   178→      if (isTypingTarget(e.target)) return
   179→   179→      const key = e.key.toLowerCase()
   180→   180→      if (key === 'j' || e.key === 'ArrowRight') {
   181→   181→        e.preventDefault()
   182→   182→        nextLinkRef.current?.click()
   183→   183→      } else if (key === 'k' || e.key === 'ArrowLeft') {
   184→   184→        e.preventDefault()
   185→   185→        prevLinkRef.current?.click()
   186→   186→      } else if (key === 'f') {
   187→   187→        e.preventDefault()
   188→   188→        toggle(effect.id)
   189→   189→      } else if (key === 's') {
   190→   190→        e.preventDefault()
   191→   191→        toggleBundle(effect.id, opts)
   192→   192→      } else if (key === 'c') {
   193→   193→        e.preventDefault()
   194→   194→        navigator.clipboard
   195→   195→          .writeText(customizedCss)
   196→   196→          .then(() => toast.success('Copied CSS to clipboard'))
   197→   197→          .catch(() => toast.error('Copy failed — please copy manually'))
   198→   198→      } else if (key === 'b') {
   199→   199→        e.preventDefault()
   200→   200→        // Toggle the drawer so `b` works as both "open" and "close".
   201→   201→        setBundleOpen((v) => !v)
   202→   202→      }
   203→   203→    }
   204→   204→    window.addEventListener('keydown', onKey)
   205→   205→    return () => window.removeEventListener('keydown', onKey)
   206→   206→  }, [effect.id, opts, customizedCss, toggle, toggleBundle])
   207→   207→
   208→   208→  // Spotlight support: track cursor and set --fx-x / --fx-y CSS vars.
   209→   209→  React.useEffect(() => {
   210→   210→    const root = previewRef.current
   211→   211→    if (!root) return
   212→   212→    const target = root.querySelector<HTMLElement>('[data-spotlight]')
   213→   213→    if (!target) return
   214→   214→    const handler = (e: MouseEvent) => {
   215→   215→      const rect = target.getBoundingClientRect()
   216→   216→      target.style.setProperty('--fx-x', `${e.clientX - rect.left}px`)
   217→   217→      target.style.setProperty('--fx-y', `${e.clientY - rect.top}px`)
   218→   218→    }
   219→   219→    target.addEventListener('mousemove', handler)
   220→   220→    return () => target.removeEventListener('mousemove', handler)
   221→   221→  }, [effect.id])
   222→   222→
   223→   223→  const combinedSnippet = React.useMemo(
   224→   224→    () =>
   225→   225→      [
   226→   226→        '<!-- HTML -->',
   227→   227→        effect.html.trim(),
   228→   228→        '',
   229→   229→        '/* CSS */',
   230→   230→        customizedCss.trim(),
   231→   231→      ].join('\n'),
   232→   232→    [effect.html, customizedCss],
   233→   233→  )
   234→   234→
   235→   235→  async function copyShareLink() {
   236→   236→    const hash = optsToHash(opts)
   237→   237→    const url = hash
   238→   238→      ? `${window.location.origin}${window.location.pathname}#${hash}`
   239→   239→      : window.location.href
   240→   240→    try {
   241→   241→      await navigator.clipboard.writeText(url)
   242→   242→      toast.success('Shareable link copied', {
   243→   243→        description: hash
   244→   244→          ? 'Includes your customization — anyone opening it sees the same tweak.'
   245→   245→          : 'Link copied (no customization applied yet).',
   246→   246→        duration: 2400,
   247→   247→      })
   248→   248→    } catch {
   249→   249→      toast.error('Could not copy link — please copy from the address bar.')
   250→   250→    }
   251→   251→  }
   252→   252→
   253→   253→  return (
   254→   254→    <div className="mx-auto w-full max-w-7xl px-4 pb-16 pt-6 sm:px-6 lg:px-8">
   255→   255→      {/* Breadcrumb / back */}
   256→   256→      <div className="mb-5 flex items-center justify-between gap-3">
   257→   257→        <Button asChild variant="ghost" size="sm" className="gap-1.5">
   258→   258→          <Link href="/library">
   259→   259→            <ArrowLeft className="h-4 w-4" /> Back to library
   260→   260→          </Link>
   261→   261→        </Button>
   262→   262→        <div className="flex items-center gap-3 text-xs text-muted-foreground">
   263→   263→          <span className="font-mono">{effect.id}</span>
   264→   264→          {isCustomized ? (
   265→   265→            <span className="inline-flex items-center gap-0.5 rounded-full bg-primary/10 px-1.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-primary">
   266→   266→              <Sparkles className="h-2.5 w-2.5" /> Edited
   267→   267→            </span>
   268→   268→          ) : null}
   269→   269→          <CopyHistoryDropdown />
   270→   270→          <UserMenu />
   271→   271→        </div>
   272→   272→      </div>
   273→   273→
   274→   274→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-[1fr_320px]">
   275→   275→        {/* Main column */}
   276→   276→        <div className="space-y-6">
   277→   277→          <Card className="overflow-hidden border-border/60 bg-card/80 backdrop-blur">
   278→   278→            <CardHeader className="gap-2 pb-3">
   279→   279→              <div className="flex flex-wrap items-start justify-between gap-3">
   280→   280→                <div className="min-w-0">
   281→   281→                  <div className="flex flex-wrap items-center gap-2">
   282→   282→                    <h1 className="text-2xl font-bold tracking-tight sm:text-3xl">
   283→   283→                      {effect.name}
   284→   284→                    </h1>
   285→   285→                    {effect.featured ? (
   286→   286→                      <Badge className="bg-amber-500/15 text-amber-600 hover:bg-amber-500/15">
   287→   287→                        Featured
   288→   288→                      </Badge>
   289→   289→                    ) : null}
   290→   290→                  </div>
   291→   291→                  <p className="mt-1.5 max-w-2xl text-sm text-muted-foreground">
   292→   292→                    {effect.description}
   293→   293→                  </p>
   294→   294→                  <div className="mt-2 flex flex-wrap items-center gap-1.5">
   295→   295→                    <Badge variant="secondary" className="font-mono text-[10px] uppercase tracking-wider">
   296→   296→                      {effect.category}
   297→   297→                    </Badge>
   298→   298→                    {(effect.tags ?? []).map((t) => (
   299→   299→                      <Badge key={t} variant="outline" className="font-mono text-[10px] uppercase tracking-wider">
   300→   300→                        {t}
   301→   301→                      </Badge>
   302→   302→                    ))}
   303→   303→                  </div>
   304→   304→                </div>
   305→   305→                <button
   306→   306→                  type="button"
   307→   307→                  onClick={() => toggleBundle(effect.id, opts)}
   308→   308→                  aria-pressed={inBundle}
   309→   309→                  aria-label={inBundle ? 'Remove from bundle' : 'Add to bundle'}
   310→   310→                  title={inBundle ? 'Remove from bundle' : 'Add to bundle'}
   311→   311→                  className={cn(
   312→   312→                    'inline-flex h-9 w-9 shrink-0 items-center justify-center rounded-full border transition-all',
   313→   313→                    inBundle
   314→   314→                      ? 'border-primary/50 bg-primary/10 text-primary hover:bg-primary/20'
   315→   315→                      : 'border-border/60 bg-background/60 text-muted-foreground hover:border-primary/40 hover:text-primary',
   316→   316→                  )}
   317→   317→                >
   318→   318→                  {inBundle ? (
   319→   319→                    <Check className="h-4 w-4" />
   320→   320→                  ) : (
   321→   321→                    <Package className="h-4 w-4" />
   322→   322→                  )}
   323→   323→                </button>
   324→   324→                <button
   325→   325→                  type="button"
   326→   326→                  onClick={() => toggle(effect.id)}
   327→   327→                  aria-pressed={isFav}
   328→   328→                  aria-label={isFav ? 'Remove from favorites' : 'Add to favorites'}
   329→   329→                  title={isFav ? 'Remove from favorites' : 'Add to favorites'}
   330→   330→                  className={cn(
   331→   331→                    'inline-flex h-9 w-9 shrink-0 items-center justify-center rounded-full border transition-all',
   332→   332→                    isFav
   333→   333→                      ? 'border-rose-400/50 bg-rose-500/10 text-rose-500 hover:bg-rose-500/20'
   334→   334→                      : 'border-border/60 bg-background/60 text-muted-foreground hover:border-rose-400/40 hover:text-rose-500',
   335→   335→                  )}
   336→   336→                >
   337→   337→                  <Heart
   338→   338→                    className={cn(
   339→   339→                      'h-4 w-4 transition-all',
   340→   340→                      isFav && 'scale-110 fill-current',
   341→   341→                    )}
   342→   342→                  />
   343→   343→                </button>
   344→   344→                <button
   345→   345→                  type="button"
   346→   346→                  onClick={() => setBundleOpen(true)}
   347→   347→                  aria-label="Open bundle"
   348→   348→                  title="Open bundle (b)"
   349→   349→                  className="inline-flex h-9 w-9 shrink-0 items-center justify-center rounded-full border border-border/60 bg-background/60 text-muted-foreground transition-all hover:border-primary/40 hover:text-primary"
   350→   350→                >
   351→   351→                  <ExternalLink className="h-4 w-4" />
   352→   352→                </button>
   353→   353→              </div>
   354→   354→            </CardHeader>
   355→   355→
   356→   356→            <CardContent className="space-y-4 pt-0">
   357→   357→              {/* Large live preview */}
   358→   358→              <div
   359→   359→                ref={previewRef}
   360→   360→                className={cn(
   361→   361→                  'relative flex min-h-[320px] items-center justify-center overflow-hidden rounded-xl border border-border/50 p-8 sm:min-h-[400px]',
   362→   362→                  surfaceDark ? 'bg-slate-950' : effect.previewClass ?? 'bg-muted/30',
   363→   363→                  isCustomized && 'ring-1 ring-primary/20',
   364→   364→                )}
   365→   365→                dangerouslySetInnerHTML={{ __html: effect.html }}
   366→   366→              />
   367→   367→              {/* Inject this effect's CSS into the document */}
   368→   368→              <style dangerouslySetInnerHTML={{ __html: customizedCss }} />
   369→   369→
   370→   370→              {/* Tabs: Code | Customize
   371→   371→                  Controlled so we can auto-open Customize when the user
   372→   372→                  arrives via a share link with #hash customization. */}
   373→   373→              <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as 'code' | 'customize')} className="w-full">
   374→   374→                <TabsList className="grid w-full grid-cols-2">
   375→   375→                  <TabsTrigger value="code" className="gap-1.5">
   376→   376→                    <Code2 className="h-3.5 w-3.5" /> Code
   377→   377→                  </TabsTrigger>
   378→   378→                  <TabsTrigger value="customize" className="gap-1.5">
   379→   379→                    <Sparkles className="h-3.5 w-3.5" /> Customize
   380→   380→                  </TabsTrigger>
   381→   381→                </TabsList>
   382→   382→
   383→   383→                <TabsContent value="code" className="mt-3 space-y-3">
   384→   384→                  <CodeBlock
   385→   385→                    code={effect.html}
   386→   386→                    filename="markup.html"
   387→   387→                    language="html"
   388→   388→                    effect={{ id: effect.id, name: effect.name, category: effect.category }}
   389→   389→                    extraCopy={{ label: 'Copy both', text: combinedSnippet, successMessage: 'Copied HTML + CSS' }}
   390→   390→                  />
   391→   391→                  <CodeBlock
   392→   392→                    code={effect.css}
   393→   393→                    filename="styles.css"
   394→   394→                    language="css"
   395→   395→                    effect={{ id: effect.id, name: effect.name, category: effect.category }}
   396→   396→                  />
   397→   397→                </TabsContent>
   398→   398→
   399→   399→                <TabsContent value="customize" className="mt-3 space-y-4">
   400→   400→                  <CustomizePanel
   401→   401→                    opts={opts}
   402→   402→                    onChange={setOpts}
   403→   403→                    onReset={() => setOpts(DEFAULT_CUSTOMIZATION)}
   404→   404→                    onPreset={(p) => setOpts({ ...p.opts })}
   405→   405→                    activePreset={activePreset}
   406→   406→                    isCustomized={isCustomized}
   407→   407→                    customizedCss={customizedCss}
   408→   408→                    combinedSnippet={combinedSnippet}
   409→   409→                    onCopyShareLink={copyShareLink}
   410→   410→                  />
   411→   411→                </TabsContent>
   412→   412→              </Tabs>
   413→   413→            </CardContent>
   414→   414→          </Card>
   415→   415→
   416→   416→          {/* Prev / next navigation */}
   417→   417→          <nav className="grid grid-cols-2 gap-3">
   418→   418→            {prev ? (
   419→   419→              <Link
   420→   420→                ref={prevLinkRef}
   421→   421→                href={`/effect/${prev.id}`}
   422→   422→                className="group flex items-center gap-3 rounded-lg border border-border/60 bg-card/60 p-3 transition-all hover:border-primary/40 hover:bg-card"
   423→   423→              >
   424→   424→                <ChevronLeft className="h-5 w-5 shrink-0 text-muted-foreground transition-colors group-hover:text-primary" />
   425→   425→                <div className="min-w-0">
   426→   426→                  <div className="text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   427→   427→                    Previous
   428→   428→                  </div>
   429→   429→                  <div className="truncate text-sm font-semibold">{prev.name}</div>
   430→   430→                </div>
   431→   431→              </Link>
   432→   432→            ) : (
   433→   433→              <div className="rounded-lg border border-dashed border-border/40 p-3 opacity-50">
   434→   434→                <div className="text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   435→   435→                  Previous
   436→   436→                </div>
   437→   437→                <div className="text-sm text-muted-foreground">Start of catalog</div>

   516→   516→/* ============================================================
   517→   517→ *  Mini preview for the similar-effects rail.
   518→   518→ *  Injects the effect's CSS scoped under a unique wrapper class
   519→   519→ *  so multiple similar effects can render side by side.
   520→   520→ * ========================================================== */
   521→   521→
   522→   522→let previewSeq = 0
   523→   523→
   524→   524→function SimilarPreview({ effect }: { effect: Effect }) {
   525→   525→  const [wrapId] = React.useState(() => `fx-preview-${++previewSeq}`)
   526→   526→  // Scope the CSS by prefixing every selector with our wrapper class.
   527→   527→  // Cheap & cheerful — works for the simple class-based selectors we generate.
   528→   528→  const scopedCss = React.useMemo(() => {
   529→   529→    // Replace each selector group `selector1, selector2 {` with
   530→   530→    // `.wrapId selector1, .wrapId selector2 {`
   531→   531→    return effect.css.replace(/(^|\})\s*([^{}]+)\{/g, (_m, brace, selectors) => {
   532→   532→      const scoped = selectors
   533→   533→        .split(',')
   534→   534→        .map((s: string) => `.${wrapId} ${s.trim()}`)
   535→   535→        .join(', ')
   536→   536→      return `${brace} ${scoped} {`
   537→   537→    })
   538→   538→  }, [effect.css, wrapId])
   539→   539→
   540→   540→  return (
   541→   541→    <div
   542→   542→      className={cn(
   543→   543→        'flex h-10 w-10 shrink-0 items-center justify-center overflow-hidden rounded-md border border-border/40',
   544→   544→        effect.darkSurface ? 'bg-slate-950' : 'bg-muted/40',
   545→   545→      )}
   546→   546→    >
   547→   547→      <style dangerouslySetInnerHTML={{ __html: scopedCss }} />
   548→   548→      <div
   549→   549→        className={wrapId}
   550→   550→        style={{ transform: 'scale(0.45)', transformOrigin: 'center' }}
   551→   551→        dangerouslySetInnerHTML={{ __html: effect.html }}
   552→   552→      />
   553→   553→    </div>
   554→   554→  )
   555→   555→}
