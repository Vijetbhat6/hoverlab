    54→/**
    55→ * Full-page detail view for a single effect.
    56→ *
    57→ * - Large live preview (always visible, customizable in real time)
    58→ * - Customize tab with hue / size / speed sliders
    59→ * - Code tab with HTML + CSS (and "Copy both")

    64→export function EffectDetail({ effect, similar, prev, next }: EffectDetailProps) {
    65→  const surfaceDark = effect.darkSurface
    66→  const previewRef = React.useRef<HTMLDivElement>(null)
    67→  const { has, toggle } = useFavorites()
    68→  const isFav = has(effect.id)
    69→  const { has: hasBundle, toggle: toggleBundle } = useBundle()
    70→  const inBundle = hasBundle(effect.id)
    71→  const [bundleOpen, setBundleOpen] = React.useState(false)
    72→
    73→  /* Active tab state ---------------------------------------------------
    74→   * We control the Tabs (value + onValueChange) instead of using
    75→   * defaultValue so we can auto-open the Customize tab when the user
    76→   * arrives via a share link that contains customization in the URL
    77→   * hash (e.g. /effect/btn-gradient#hue=15&sat=25). On subsequent
    78→   * prev/next navigations the user's current tab is preserved — only
    79→   * the initial mount checks the hash.
    80→   *
    81→   * We intentionally start with 'code' on both server and client
    82→   * (avoids a hydration mismatch — the server can't see the URL hash)
    83→   * and flip to 'customize' in a useEffect if the hash has any
    84→   * customization params.
    85→   */
    86→  const [activeTab, setActiveTab] = React.useState<'code' | 'customize'>('code')
    87→  React.useEffect(() => {
    88→    const parts = parseHash(window.location.hash)
    89→    const hasCustomization =
    90→      parts.hue !== undefined ||
    91→      parts.saturation !== undefined ||
    92→      parts.scale !== undefined ||
    93→      parts.speed !== undefined
    94→    if (hasCustomization) setActiveTab('customize')
    95→    // Only run on mount — prev/next nav preserves the user's tab.
    96→  }, [])
    97→
    98→  /* Customization state -----------------------------------------------
    99→   * Initialize from the URL hash on first mount, so a shared link
   100→   * (e.g. /effect/btn-gradient#hue=15&sat=25) restores the exact
   101→   * customized view. When the user navigates to a different effect
   102→   * via prev/next, we re-read the hash for that URL too.
   103→   */
   104→  const [opts, setOpts] = React.useState<CustomizationOptions>(() =>
   105→    typeof window === 'undefined'
   106→      ? DEFAULT_CUSTOMIZATION
   107→      : hashToOpts(parseHash(window.location.hash)),
   108→  )
   109→  // Flag used to suppress the very first hash-write effect (so we don't
   110→  // re-write the hash that the user just landed on, which would trigger
   111→  // an unnecessary history entry).
   112→  const skipHashWriteRef = React.useRef(true)
   113→
   114→  React.useEffect(() => {
   115→    // When effect.id changes (prev/next navigation via internal Link),
   116→    // re-read the hash because Next.js client-side navigation does NOT
   117→    // re-mount this component — it just updates props.
   118→    const next = hashToOpts(parseHash(window.location.hash))
   119→    setOpts(next)
   120→    skipHashWriteRef.current = true
   121→  }, [effect.id])
   122→
   123→  // Write opts to the URL hash whenever they change (except on the very
   124→  // first run after a navigation, to avoid a redundant history entry).
   125→  React.useEffect(() => {
   126→    if (skipHashWriteRef.current) {
   127→      skipHashWriteRef.current = false
   128→      return
   129→    }
   130→    const hash = optsToHash(opts)
   131→    const target = hash ? `#${hash}` : ''
   132→    if (window.location.hash !== target) {
   133→      window.history.replaceState(null, '', window.location.pathname + window.location.search + target)
   134→    }
   135→  }, [opts])
   136→
   137→  // Listen for hashchange events (back/forward, manual URL edits) and
   138→  // sync opts from the URL. This makes the share-link round-trip work.
   139→  React.useEffect(() => {
   140→    const onHashChange = () => {
   141→      const next = hashToOpts(parseHash(window.location.hash))
   142→      skipHashWriteRef.current = true
   143→      setOpts(next)
   144→    }
   145→    window.addEventListener('hashchange', onHashChange)
   146→    return () => window.removeEventListener('hashchange', onHashChange)
   147→  }, [])
   148→
   149→  const customizedCss = React.useMemo(
   150→    () => customizeCss(effect.css, opts),
   151→    [effect.css, opts],
   152→  )
   153→  const isCustomized = customizedCss !== effect.css
   154→  const activePreset = matchingPreset(opts)
   155→
   156→  // Detail-page keyboard shortcuts:
   157→  //   j / →  → next effect
   158→  //   k / ←  → previous effect
   159→  //   f      → toggle favorite
   160→  //   s      → toggle bundle (saves current customization)
   161→  //   c      → copy current (customized) CSS
   162→  //   b      → open the bundle drawer
   163→  //   Esc    → close any open dialog (handled by Radix)
   164→  // We rely on Next.js <Link> for prev/next so client-side navigation
   165→  // preserves state. Pressing j/k triggers a click on the corresponding
   166→  // link element via ref.
   167→  const prevLinkRef = React.useRef<HTMLAnchorElement>(null)
   168→  const nextLinkRef = React.useRef<HTMLAnchorElement>(null)
   169→
   170→  React.useEffect(() => {
   171→    function isTypingTarget(t: EventTarget | null): boolean {
   172→      if (!(t instanceof HTMLElement)) return false
   173→      const tag = t.tagName
   174→      return tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || t.isContentEditable
   175→    }
   176→    function onKey(e: KeyboardEvent) {
   177→      if (e.metaKey || e.ctrlKey || e.altKey) return
   178→      if (isTypingTarget(e.target)) return
   179→      const key = e.key.toLowerCase()
   180→      if (key === 'j' || e.key === 'ArrowRight') {
   181→        e.preventDefault()
   182→        nextLinkRef.current?.click()
   183→      } else if (key === 'k' || e.key === 'ArrowLeft') {
   184→        e.preventDefault()
   185→        prevLinkRef.current?.click()
   186→      } else if (key === 'f') {
   187→        e.preventDefault()
   188→        toggle(effect.id)
   189→      } else if (key === 's') {
   190→        e.preventDefault()
   191→        toggleBundle(effect.id, opts)
   192→      } else if (key === 'c') {
   193→        e.preventDefault()
   194→        navigator.clipboard
   195→          .writeText(customizedCss)
   196→          .then(() => toast.success('Copied CSS to clipboard'))
   197→          .catch(() => toast.error('Copy failed — please copy manually'))
   198→      } else if (key === 'b') {
   199→        e.preventDefault()
   200→        // Toggle the drawer so `b` works as both "open" and "close".
   201→        setBundleOpen((v) => !v)
   202→      }
   203→    }
   204→    window.addEventListener('keydown', onKey)
   205→    return () => window.removeEventListener('keydown', onKey)
   206→  }, [effect.id, opts, customizedCss, toggle, toggleBundle])
   207→
   208→  // Spotlight support: track cursor and set --fx-x / --fx-y CSS vars.
   209→  React.useEffect(() => {
   210→    const root = previewRef.current
   211→    if (!root) return
   212→    const target = root.querySelector<HTMLElement>('[data-spotlight]')
   213→    if (!target) return
   214→    const handler = (e: MouseEvent) => {
   215→      const rect = target.getBoundingClientRect()
   216→      target.style.setProperty('--fx-x', `${e.clientX - rect.left}px`)
   217→      target.style.setProperty('--fx-y', `${e.clientY - rect.top}px`)
   218→    }
   219→    target.addEventListener('mousemove', handler)
   220→    return () => target.removeEventListener('mousemove', handler)
   221→  }, [effect.id])
   222→
   223→  const combinedSnippet = React.useMemo(
   224→    () =>
   225→      [
   226→        '<!-- HTML -->',
   227→        effect.html.trim(),
   228→        '',
   229→        '/* CSS */',
   230→        customizedCss.trim(),
   231→      ].join('\n'),
   232→    [effect.html, customizedCss],
   233→  )
   234→
   235→  async function copyShareLink() {
   236→    const hash = optsToHash(opts)
   237→    const url = hash
   238→      ? `${window.location.origin}${window.location.pathname}#${hash}`
   239→      : window.location.href
   240→    try {
   241→      await navigator.clipboard.writeText(url)
   242→      toast.success('Shareable link copied', {
   243→        description: hash
   244→          ? 'Includes your customization — anyone opening it sees the same tweak.'
   245→          : 'Link copied (no customization applied yet).',
   246→        duration: 2400,
   247→      })
   248→    } catch {
   249→      toast.error('Could not copy link — please copy from the address bar.')
   250→    }
   251→  }
   252→
   253→  return (
   254→    <div className="mx-auto w-full max-w-7xl px-4 pb-16 pt-6 sm:px-6 lg:px-8">
   255→      {/* Breadcrumb / back */}
   256→      <div className="mb-5 flex items-center justify-between gap-3">
   257→        <Button asChild variant="ghost" size="sm" className="gap-1.5">
   258→          <Link href="/library">
   259→            <ArrowLeft className="h-4 w-4" /> Back to library
   260→          </Link>
   261→        </Button>
   262→        <div className="flex items-center gap-3 text-xs text-muted-foreground">
   263→          <span className="font-mono">{effect.id}</span>
   264→          {isCustomized ? (
   265→            <span className="inline-flex items-center gap-0.5 rounded-full bg-primary/10 px-1.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-primary">
   266→              <Sparkles className="h-2.5 w-2.5" /> Edited
   267→            </span>
   268→          ) : null}
   269→          <CopyHistoryDropdown />
   270→          <UserMenu />
   271→        </div>
   272→      </div>
   273→
   274→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-[1fr_320px]">
   275→        {/* Main column */}
   276→        <div className="space-y-6">
   277→          <Card className="overflow-hidden border-border/60 bg-card/80 backdrop-blur">
   278→            <CardHeader className="gap-2 pb-3">
   279→              <div className="flex flex-wrap items-start justify-between gap-3">
   280→                <div className="min-w-0">
   281→                  <div className="flex flex-wrap items-center gap-2">
   282→                    <h1 className="text-2xl font-bold tracking-tight sm:text-3xl">
   283→                      {effect.name}
   284→                    </h1>
   285→                    {effect.featured ? (
   286→                      <Badge className="bg-amber-500/15 text-amber-600 hover:bg-amber-500/15">
   287→                        Featured
   288→                      </Badge>
   289→                    ) : null}
   290→                  </div>
   291→                  <p className="mt-1.5 max-w-2xl text-sm text-muted-foreground">
   292→                    {effect.description}
   293→                  </p>
   294→                  <div className="mt-2 flex flex-wrap items-center gap-1.5">
   295→                    <Badge variant="secondary" className="font-mono text-[10px] uppercase tracking-wider">
   296→                      {effect.category}
   297→                    </Badge>
   298→                    {(effect.tags ?? []).map((t) => (
   299→                      <Badge key={t} variant="outline" className="font-mono text-[10px] uppercase tracking-wider">
   300→                        {t}
   301→                      </Badge>
   302→                    ))}
   303→                  </div>
   304→                </div>
   305→                <button
   306→                  type="button"
   307→                  onClick={() => toggleBundle(effect.id, opts)}
   308→                  aria-pressed={inBundle}
   309→                  aria-label={inBundle ? 'Remove from bundle' : 'Add to bundle'}
   310→                  title={inBundle ? 'Remove from bundle' : 'Add to bundle'}
   311→                  className={cn(
   312→                    'inline-flex h-9 w-9 shrink-0 items-center justify-center rounded-full border transition-all',
   313→                    inBundle
   314→                      ? 'border-primary/50 bg-primary/10 text-primary hover:bg-primary/20'
   315→                      : 'border-border/60 bg-background/60 text-muted-foreground hover:border-primary/40 hover:text-primary',
   316→                  )}
   317→                >
   318→                  {inBundle ? (
   319→                    <Check className="h-4 w-4" />
   320→                  ) : (
   321→                    <Package className="h-4 w-4" />
   322→                  )}
   323→                </button>
   324→                <button
   325→                  type="button"
   326→                  onClick={() => toggle(effect.id)}
   327→                  aria-pressed={isFav}
   328→                  aria-label={isFav ? 'Remove from favorites' : 'Add to favorites'}
   329→                  title={isFav ? 'Remove from favorites' : 'Add to favorites'}
   330→                  className={cn(
   331→                    'inline-flex h-9 w-9 shrink-0 items-center justify-center rounded-full border transition-all',
   332→                    isFav
   333→                      ? 'border-rose-400/50 bg-rose-500/10 text-rose-500 hover:bg-rose-500/20'
   334→                      : 'border-border/60 bg-background/60 text-muted-foreground hover:border-rose-400/40 hover:text-rose-500',
   335→                  )}
   336→                >
   337→                  <Heart
   338→                    className={cn(
   339→                      'h-4 w-4 transition-all',
   340→                      isFav && 'scale-110 fill-current',
   341→                    )}
   342→                  />
   343→                </button>
   344→                <button
   345→                  type="button"
   346→                  onClick={() => setBundleOpen(true)}
   347→                  aria-label="Open bundle"
   348→                  title="Open bundle (b)"
   349→                  className="inline-flex h-9 w-9 shrink-0 items-center justify-center rounded-full border border-border/60 bg-background/60 text-muted-foreground transition-all hover:border-primary/40 hover:text-primary"
   350→                >
   351→                  <ExternalLink className="h-4 w-4" />
   352→                </button>
   353→              </div>
   354→            </CardHeader>
   355→
   356→            <CardContent className="space-y-4 pt-0">
   357→              {/* Large live preview */}
   358→              <div
   359→                ref={previewRef}
   360→                className={cn(
   361→                  'relative flex min-h-[320px] items-center justify-center overflow-hidden rounded-xl border border-border/50 p-8 sm:min-h-[400px]',
   362→                  surfaceDark ? 'bg-slate-950' : effect.previewClass ?? 'bg-muted/30',
   363→                  isCustomized && 'ring-1 ring-primary/20',
   364→                )}
   365→                dangerouslySetInnerHTML={{ __html: effect.html }}
   366→              />
   367→              {/* Inject this effect's CSS into the document */}
   368→              <style dangerouslySetInnerHTML={{ __html: customizedCss }} />
   369→
   370→              {/* Tabs: Code | Customize
   371→                  Controlled so we can auto-open Customize when the user
   372→                  arrives via a share link with #hash customization. */}
   373→              <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as 'code' | 'customize')} className="w-full">
   374→                <TabsList className="grid w-full grid-cols-2">
   375→                  <TabsTrigger value="code" className="gap-1.5">
   376→                    <Code2 className="h-3.5 w-3.5" /> Code
   377→                  </TabsTrigger>
   378→                  <TabsTrigger value="customize" className="gap-1.5">
   379→                    <Sparkles className="h-3.5 w-3.5" /> Customize
   380→                  </TabsTrigger>
   381→                </TabsList>
   382→
   383→                <TabsContent value="code" className="mt-3 space-y-3">
   384→                  <CodeBlock
   385→                    code={effect.html}
   386→                    filename="markup.html"
   387→                    language="html"
   388→                    effect={{ id: effect.id, name: effect.name, category: effect.category }}
   389→                    extraCopy={{ label: 'Copy both', text: combinedSnippet, successMessage: 'Copied HTML + CSS' }}
   390→                  />
   391→                  <CodeBlock
   392→                    code={effect.css}
   393→                    filename="styles.css"
   394→                    language="css"
   395→                    effect={{ id: effect.id, name: effect.name, category: effect.category }}
   396→                  />
   397→                </TabsContent>
   398→
   399→                <TabsContent value="customize" className="mt-3 space-y-4">
   400→                  <CustomizePanel
   401→                    opts={opts}
   402→                    onChange={setOpts}
   403→                    onReset={() => setOpts(DEFAULT_CUSTOMIZATION)}
   404→                    onPreset={(p) => setOpts({ ...p.opts })}
   405→                    activePreset={activePreset}
   406→                    isCustomized={isCustomized}
   407→                    customizedCss={customizedCss}
   408→                    combinedSnippet={combinedSnippet}
   409→                    onCopyShareLink={copyShareLink}
   410→                  />
   411→                </TabsContent>
   412→              </Tabs>
   413→            </CardContent>
   414→          </Card>
   415→
   416→          {/* Prev / next navigation */}
   417→          <nav className="grid grid-cols-2 gap-3">
   418→            {prev ? (
   419→              <Link
   420→                ref={prevLinkRef}
   421→                href={`/effect/${prev.id}`}
   422→                className="group flex items-center gap-3 rounded-lg border border-border/60 bg-card/60 p-3 transition-all hover:border-primary/40 hover:bg-card"
   423→              >
   424→                <ChevronLeft className="h-5 w-5 shrink-0 text-muted-foreground transition-colors group-hover:text-primary" />
   425→                <div className="min-w-0">
   426→                  <div className="text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   427→                    Previous
   428→                  </div>
   429→                  <div className="truncate text-sm font-semibold">{prev.name}</div>
   430→                </div>
   431→              </Link>
   432→            ) : (
   433→              <div className="rounded-lg border border-dashed border-border/40 p-3 opacity-50">
   434→                <div className="text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   435→                  Previous
   436→                </div>
   437→                <div className="text-sm text-muted-foreground">Start of catalog</div>
   438→              </div>
   439→            )}
   440→            {next ? (
   441→              <Link
   442→                ref={nextLinkRef}
   443→                href={`/effect/${next.id}`}
   444→                className="group flex items-center justify-end gap-3 rounded-lg border border-border/60 bg-card/60 p-3 text-right transition-all hover:border-primary/40 hover:bg-card"
   445→              >
   446→                <div className="min-w-0">
   447→                  <div className="text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   448→                    Next
   449→                  </div>
   450→                  <div className="truncate text-sm font-semibold">{next.name}</div>
   451→                </div>
   452→                <ChevronRight className="h-5 w-5 shrink-0 text-muted-foreground transition-colors group-hover:text-primary" />
   453→              </Link>
   454→            ) : (
   455→              <div className="rounded-lg border border-dashed border-border/40 p-3 text-right opacity-50">
   456→                <div className="text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   457→                  Next
   458→                </div>
   459→                <div className="text-sm text-muted-foreground">End of catalog</div>
   460→              </div>
   461→            )}
   462→          </nav>
   463→        </div>
   464→
   465→        {/* Sidebar: similar effects */}
   466→        <aside className="space-y-4 lg:sticky lg:top-20 lg:self-start">
   467→          <div className="rounded-lg border border-border/60 bg-card/60 p-4">
   468→            <div className="mb-3 flex items-center justify-between">
   469→              <h2 className="text-sm font-semibold tracking-tight">Similar effects</h2>
   470→              <Badge variant="outline" className="text-[10px]">
   471→                {effect.category}
   472→              </Badge>
   473→            </div>
   474→            {similar.length === 0 ? (
   475→              <p className="text-xs text-muted-foreground">
   476→                No other effects in this category yet.
   477→              </p>
   478→            ) : (
   479→              <ul className="space-y-2">
   480→                {similar.map((s) => (
   481→                  <li key={s.id}>
   482→                    <Link
   483→                      href={`/effect/${s.id}`}
   484→                      className="group flex items-center gap-3 rounded-md p-2 transition-colors hover:bg-muted/60"
   485→                    >
   486→                      <SimilarPreview effect={s} />
   487→                      <div className="min-w-0 flex-1">

   489→                          {s.name}

   491→                        <div className="truncate text-[11px] text-muted-foreground">
   492→                          {s.description}
   493→                        </div>
   494→                      </div>
   495→                      <ExternalLink className="h-3 w-3 shrink-0 text-muted-foreground opacity-0 transition-opacity group-hover:opacity-100" />
   496→                    </Link>
   497→                  </li>
   498→                ))}
   499→              </ul>
   500→            )}
   501→            <Button asChild variant="outline" size="sm" className="mt-3 w-full">
   502→              <Link href={`/library?filter=${encodeURIComponent(effect.category)}`}>
   503→                Browse all {effect.category.toLowerCase()}
   504→              </Link>
   505→            </Button>
   506→          </div>
   507→        </aside>
   508→      </div>
   509→
   510→      {/* Slide-out bundle drawer (opened via 'b' shortcut or the bundle button) */}
   511→      <BundleDrawer open={bundleOpen} onOpenChange={setBundleOpen} />
   512→    </div>
   513→  )
   514→}

   516→/* ============================================================
   517→ *  Mini preview for the similar-effects rail.
   518→ *  Injects the effect's CSS scoped under a unique wrapper class
   519→ *  so multiple similar effects can render side by side.
   520→ * ========================================================== */
   521→
   522→let previewSeq = 0
   523→
   524→function SimilarPreview({ effect }: { effect: Effect }) {
   525→  const [wrapId] = React.useState(() => `fx-preview-${++previewSeq}`)
   526→  // Scope the CSS by prefixing every selector with our wrapper class.
   527→  // Cheap & cheerful — works for the simple class-based selectors we generate.
   528→  const scopedCss = React.useMemo(() => {
   529→    // Replace each selector group `selector1, selector2 {` with
   530→    // `.wrapId selector1, .wrapId selector2 {`
   531→    return effect.css.replace(/(^|\})\s*([^{}]+)\{/g, (_m, brace, selectors) => {
   532→      const scoped = selectors
   533→        .split(',')
   534→        .map((s: string) => `.${wrapId} ${s.trim()}`)
   535→        .join(', ')
   536→      return `${brace} ${scoped} {`
   537→    })
   538→  }, [effect.css, wrapId])
   539→
   540→  return (
   541→    <div
   542→      className={cn(
   543→        'flex h-10 w-10 shrink-0 items-center justify-center overflow-hidden rounded-md border border-border/40',
   544→        effect.darkSurface ? 'bg-slate-950' : 'bg-muted/40',
   545→      )}
   546→    >
   547→      <style dangerouslySetInnerHTML={{ __html: scopedCss }} />
   548→      <div
   549→        className={wrapId}
   550→        style={{ transform: 'scale(0.45)', transformOrigin: 'center' }}
   551→        dangerouslySetInnerHTML={{ __html: effect.html }}
   552→      />
   553→    </div>
   554→  )
   555→}

   736→interface SliderRowProps {
   737→  label: string
   738→  value: number
   739→  min: number
   740→  max: number
   741→  step: number
   742→  unit: string
   743→  format?: (v: number) => string

   748→function SliderRow({
   749→  label,
   750→  value,
   751→  min,
   752→  max,
   753→  step,
   754→  unit,
   755→  format,
   756→  description,

   768→      <Slider
   769→        value={[value]}
   770→        min={min}
   771→        max={max}
   772→        step={step}
   773→        onValueChange={(arr) => arr[0] !== undefined && onChange(arr[0])}
   774→        className="w-full"
   775→      />
   776→      {description ? (
   777→        <p className="text-[11px] text-muted-foreground/80">{description}</p>
   778→      ) : null}
   779→    </div>
   780→  )
